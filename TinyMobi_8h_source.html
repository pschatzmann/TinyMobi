1/*
2 * TinyMobi - header-only MOBI / PalmDOC parser for Arduino and host C++.
3 *
4 * Designed for document readers on memory constrained MCUs:
5 * - The file is never loaded completely. The PalmDB record table, the
6 * headers and the metadata are read on demand through a small source
7 * adapter (memory buffer or any seekable file, e.g. SD / LittleFS / FFat).
8 * - Text is decompressed one record ("chunk") at a time into a single
9 * fixed size buffer (RecordBufferSize, default 4 KB). No heap is used.
10 * - Text can be streamed sequentially through read(), or random accessed
11 * per section through seek(), which makes page caching and
12 * bookmarks cheap.
13 * - Output is always UTF-8. CP1252 books are converted on the fly.
14 * - Optional plain text mode strips markup and decodes HTML entities.
15 *
16 * Supported: PalmDOC (LZ77), HUFF/CDIC and uncompressed text, trailing data
17 * entries, EXTH metadata (title, author, publisher, language, date, cover),
18 * the table of contents (NCX index), raw access to image records with image
19 * type detection, UTF-8 and CP1252 text encodings.
20 * KF8 (AZW3 and hybrid files): the parts are assembled from the skeleton and
21 * fragment tables and read like any other text. Image references (recindex and
22 * kindle:embed) are found with forEachImageRef().
23 * Not supported: DRM encryption (reported as Error::Encrypted). The FDST
24 * flow table is not parsed; the text is assumed to start at offset 0.
25 *
26 * Usage (memory buffer):
27 * tinymobi::MemoryMobi doc(tinymobi::MemorySource(data, len));
28 * if (doc.open() == tinymobi::Error::None) { ... doc.read(buf, n) ... }
29 *
30 * Usage (file):
31 * File f = SD.open("/book.mobi");
32 * tinymobi::FileMobi<File> doc(tinymobi::StreamSource<File>(f));
33 *
34 * License: MIT
35 */

206inline size_t utf8Encode(uint32_t cp, uint8_t* o) {
207 if (cp < 0x80) {
208 o[0] = static_cast<uint8_t>(cp);
209 return 1;
210 }
211 if (cp < 0x800) {
212 o[0] = static_cast<uint8_t>(0xC0 | (cp >> 6));
213 o[1] = static_cast<uint8_t>(0x80 | (cp & 0x3F));
214 return 2;
215 }
216 if (cp < 0x10000) {
217 o[0] = static_cast<uint8_t>(0xE0 | (cp >> 12));
218 o[1] = static_cast<uint8_t>(0x80 | ((cp >> 6) & 0x3F));
219 o[2] = static_cast<uint8_t>(0x80 | (cp & 0x3F));
220 return 3;
221 }
222 o[0] = static_cast<uint8_t>(0xF0 | (cp >> 18));
223 o[1] = static_cast<uint8_t>(0x80 | ((cp >> 12) & 0x3F));
224 o[2] = static_cast<uint8_t>(0x80 | ((cp >> 6) & 0x3F));
225 o[3] = static_cast<uint8_t>(0x80 | (cp & 0x3F));
226 return 4;
227}

230inline size_t trimIncompleteUtf8(const char* s, size_t o) {
231 size_t i = o;
232 for (int back = 0; i > 0 && back < 4; ++back) {
233 --i;
234 uint8_t c = static_cast<uint8_t>(s[i]);
235 if ((c & 0xC0) != 0x80) { // lead byte or ASCII
236 size_t need = c >= 0xF0 ? 4 : c >= 0xE0 ? 3 : c >= 0xC0 ? 2 : 1;
237 return (i + need > o) ? i : o;
238 }

249inline bool lookupEntity(const uint8_t* s, size_t n, uint32_t& cp) {
250 static const NamedEntity kEntities[] = {
251 {"amp", '&'}, {"lt", '<'}, {"gt", '>'},
252 {"quot", '"'}, {"apos", '\''}, {"nbsp", 0xA0},
253 {"copy", 0xA9}, {"reg", 0xAE}, {"deg", 0xB0},
254 {"middot", 0xB7}, {"laquo", 0xAB}, {"raquo", 0xBB},
255 {"times", 0xD7}, {"szlig", 0xDF}, {"aacute", 0xE1},
256 {"agrave", 0xE0}, {"auml", 0xE4}, {"eacute", 0xE9},
257 {"egrave", 0xE8}, {"ouml", 0xF6}, {"uuml", 0xFC},
258 {"ndash", 0x2013}, {"mdash", 0x2014}, {"lsquo", 0x2018},
259 {"rsquo", 0x2019}, {"ldquo", 0x201C}, {"rdquo", 0x201D},
260 {"bull", 0x2022}, {"hellip", 0x2026}, {"euro", 0x20AC},
261 };
262 if (n == 0) return false;
263 if (s[0] == '#') { // numeric reference: &#NNN; or &#xHH;
264 size_t i = 1;
265 uint32_t base = 10;
266 if (n > 1 && (s[1] == 'x' || s[1] == 'X')) {
267 base = 16;
268 i = 2;
269 }
270 if (i >= n) return false;
271 uint32_t v = 0;
272 for (; i < n; ++i) {
273 uint8_t c = s[i];
274 uint32_t d;
275 if (c >= '0' && c <= '9') d = c - '0';
276 else if (base == 16 && c >= 'a' && c <= 'f') d = c - 'a' + 10;
277 else if (base == 16 && c >= 'A' && c <= 'F') d = c - 'A' + 10;
278 else return false;
279 v = v * base + d;
280 if (v > 0x10FFFF) return false;
281 }
282 if (v == 0) return false;
283 cp = v;
284 return true;
285 }
286 for (const NamedEntity& e : kEntities) {
287 size_t len = strlen(e.name);
288 if (len == n && memcmp(e.name, s, n) == 0) {
289 cp = e.cp;
290 return true;
291 }
292 }
293 return false;
294}

382 Error open() {
383 close();
384 _total = _src.size();
385 uint8_t head[78];
386 if (_total < 78 || _src.read(0, head, sizeof(head)) != sizeof(head)) {
387 return fail(Error::NotPalmDb);
388 }
389 if (memcmp(head + 64, "MOBI", 4) != 0) return fail(Error::NotMobi);
390 _numRecords = detail::be16(head + 76);
391 if (_numRecords == 0) return fail(Error::NotMobi);
392 _info.recordCount = _numRecords;
393
394 // record 0 holds the PalmDOC and MOBI headers
395 _rec0 = recordOffset(0);
396 uint32_t end0 = recordEnd(0);
397 if (_rec0 == kNoRecord || end0 == kNoRecord || end0 > _total || end0 <= _rec0) {
398 return fail(Error::CorruptData);
399 }
400 _rec0Len = end0 - _rec0;
401
402 uint8_t magic[4];
403 if (!readRel(16, magic, 4) || memcmp(magic, "MOBI", 4) != 0) {
404 return fail(Error::NotMobi);
405 }
406 uint32_t headerLen = 0;
407 if (!rd32(20, headerLen)) return fail(Error::CorruptData);
408 _mobiHeaderLen = headerLen;
409
410 uint32_t comp = 0, textLen = 0, recCount = 0, recSize = 0, enc = 0;
411 if (!rd16(0, comp) || !rd32(4, textLen) || !rd16(8, recCount) ||
412 !rd16(10, recSize) || !rd16(12, enc)) {
413 return fail(Error::CorruptData);
414 }
415 if (enc != 0) return fail(Error::Encrypted);
416 if (comp != 1 && comp != 2 && comp != kHuffCompression) {
417 return fail(Error::UnsupportedCompression);
418 }
419
420 _info = Info();
421 memset(&_meta, 0, sizeof(_meta));
422 _info.recordCount = _numRecords;
423 _info.compression = static_cast<uint16_t>(comp);
424 _info.textLength = textLen;
425 _info.recordSize = static_cast<uint16_t>(recSize);
426 _info.textChunks = recCount < _numRecords - 1 ? recCount : _numRecords - 1;
427 _info.firstImageRecord = kNoRecord;
428 _info.coverRecord = kNoRecord;
429
430 uint32_t encoding = 0;
431 if (mobiField(28, encoding)) _utf8 = (encoding == 65001);
432 _info.utf8 = _utf8;
433
434 uint32_t version = 0;
435 if (mobiField(0x68, version)) {
436 _info.mobiVersion = version;
437 _info.isKf8 = version >= 8;
438 }
439
440 uint32_t firstImage = 0;
441 if (mobiField(0x6C, firstImage) && firstImage != 0 && firstImage < _numRecords) {
442 _info.firstImageRecord = firstImage;
443 }
444
445 uint32_t extra = 0;
446 if (headerLen + 16 >= 0xF4 && rd16(0xF2, extra)) {
447 _info.extraFlags = static_cast<uint16_t>(extra);
448 }
449
450 if (comp == kHuffCompression) {
451 uint32_t huffRec = 0, huffCount = 0;
452 if (!mobiField(0x70, huffRec) || !mobiField(0x74, huffCount)) {
453 return fail(Error::CorruptData);
454 }
455 Error he = loadHuffTables(huffRec, huffCount);
456 if (he != Error::None) return fail(he);
457 }
458
459 // full title, located in record 0
460 uint32_t nameOff = 0, nameLen = 0;
461 if (mobiField(0x54, nameOff) && mobiField(0x58, nameLen) && nameLen > 0 &&
462 nameOff < _rec0Len && nameLen <= _rec0Len - nameOff) {
463 copyText(_rec0 + nameOff, nameLen, _meta.title, sizeof(_meta.title));
464 }
465
466 uint32_t exthFlags = 0;
467 if (mobiField(0x80, exthFlags) && (exthFlags & 0x40) != 0) {
468 parseExth();
469 }
470
471 _mobi6View.firstRec = 1;
472 _mobi6View.chunks = _info.textChunks;
473 _mobi6View.recordSize = _info.recordSize;
474 _mobi6View.textLength = _info.textLength;
475 _mobi6View.compression = _info.compression;
476 _mobi6View.extraFlags = _info.extraFlags;
477 _kf8Loaded = false;
478 _partOpen = false;
479 _walkSection = 0;
480 _walkStart = 0;
481 _rawChunk = kNoRecord;
482
483 _open = true;
484 // KF8 text is used when the book has the KF8 parts (AZW3 and hybrid books)
485 _useKf8 = ensureKf8() == Error::None && _kf8Skel != kNoRecord && _kf8Frag != kNoRecord;
486 _error = Error::None;
487 rewind();
488 return Error::None;
489 }

545 template <typename Fn>
546 Error forEachImageRef(Fn fn) {
547 if (!_open) return fail(Error::NotOpen);
548 Position saved = position();
549 Error e = rewind();
550 if (e != Error::None) return e;
551 uint8_t tag[256];
552 size_t tagLen = 0;
553 bool inTag = false;
554 bool stop = false;
555 Position tagStart;
556 // the buffer loaded by rewind() is scanned too, then each following one
557 for (bool more = true; more && !stop; more = loadNext()) {
558 for (uint32_t k = 0; k < _recLen && !stop; ++k) {
559 uint8_t b = _rec[k];
560 if (!inTag) {
561 if (b == '<') {
562 inTag = true;
563 tagLen = 0;
564 tagStart = Position(_curChunk, _secBase + k);
565 }
566 continue;
567 }
568 if (b == '>') {
569 inTag = false;
570 uint32_t number = 0;
571 if (parseImageTag(tag, tagLen, number)) {
572 ImageRef ref;
573 ref.position = tagStart;
574 ref.record = imageRecordOf(number);
575 if (!fn(ref)) stop = true;
576 }
577 continue;
578 }
579 if (tagLen < sizeof(tag)) {
580 tag[tagLen++] = b;
581 } else {
582 inTag = false; // too long to be an image tag
583 }
584 }
585 }
586 seek(saved); // the reading position is kept
587 return Error::None;
588 }

711 bool resolveLink(const char* target, Position& out) {
712 if (!_open || target == nullptr) return false;
713 static const char kFilepos[] = "filepos=";
714 const size_t fl = sizeof(kFilepos) - 1;
715 if (strncmp(target, kFilepos, fl) == 0) {
716 uint32_t v = 0;
717 for (const char* c = target + fl; *c != 0; ++c) {
718 if (*c < '0' || *c > '9') return false;
719 v = v * 10 + static_cast<uint32_t>(*c - '0');
720 }
721 Position p = textPosition(v);
722 if (p.section == kNoRecord) return false;
723 out = p;
724 return true;
725 }
726 return resolveKf8Link(target, out);
727 }

733 size_t read(uint8_t* dst, size_t n) {
734 if (!_open || dst == nullptr) return 0;
735 size_t out = 0;
736 while (out < n) {
737 if (_pendOff < _pendLen) {
738 dst[out++] = _pend[_pendOff++];
739 continue;
740 }
741 // fast path: markup and UTF-8 book -> plain memcpy from the chunk
742 if (_mode == TextMode::Markup && _utf8) {
743 while (_recPos >= _recLen) {
744 if (!loadNext()) return out;
745 }
746 size_t avail = _recLen - _recPos;
747 size_t count = avail < n - out ? avail : n - out;
748 memcpy(dst + out, _rec + _recPos, count);
749 _recPos += static_cast<uint32_t>(count);
750 out += count;
751 _lastOut = dst[out - 1];
752 continue;
753 }
754 _pendOff = _pendLen = 0;
755 if (!produce()) break;
756 }
757 return out;
758 }

774 bool readAbs(uint32_t pos, void* dst, size_t n) { return _src.read(pos, dst, n) == n; }
775
776 static uint32_t countBits(uint8_t v) {
777 uint32_t n = 0;
778 for (; v != 0; v = static_cast<uint8_t>(v >> 1)) n += v & 1;
779 return n;
780 }
781
783 bool windowByte(BitWindow& w, uint32_t srcStart, uint32_t srcLen, uint32_t i, uint8_t& b) {
784 if (i < w.base || i >= w.base + w.len) {
785 uint32_t want = srcLen - i < sizeof(w.buf) ? srcLen - i : static_cast<uint32_t>(sizeof(w.buf));
786 if (!readAbs(srcStart + i, w.buf, want)) return false;
787 w.base = i;
788 w.len = want;
789 }
790 b = w.buf[i - w.base];
791 return true;
792 }
793
795 bool peekBits32(BitWindow& w, uint32_t srcStart, uint32_t srcLen, uint64_t bitPos,
796 uint32_t& code) {
797 uint32_t byteIdx = static_cast<uint32_t>(bitPos >> 3);
798 uint32_t shift = static_cast<uint32_t>(bitPos & 7);
799 uint64_t v = 0;
800 for (uint32_t k = 0; k < 5; ++k) {
801 uint8_t b = 0;
802 uint32_t i = byteIdx + k;
803 if (i < srcLen && !windowByte(w, srcStart, srcLen, i, b)) return false;
804 v = (v << 8) | b;
805 }
806 code = static_cast<uint32_t>((v >> (8 - shift)) & 0xFFFFFFFFu);
807 return true;
808 }

810 uint64_t huffMinCode(uint32_t len) const {
811 return static_cast<uint64_t>(_dict2[2 * (len - 1)]) << (32 - len);
812 }
813
814 uint64_t huffMaxCode(uint32_t len) const {
815 return ((static_cast<uint64_t>(_dict2[2 * (len - 1) + 1]) + 1) << (32 - len)) - 1;
816 }

819 bool loadBE32Table(uint32_t base, uint32_t relOff, uint32_t recLen, uint32_t* dst,
820 uint32_t count) {
821 if (relOff > recLen || count * 4 > recLen - relOff) return false;
822 uint8_t t[64];
823 for (uint32_t done = 0; done < count;) {
824 uint32_t n = count - done < 16 ? count - done : 16;
825 if (!readAbs(base + relOff + 4 * done, t, 4 * n)) return false;
826 for (uint32_t k = 0; k < n; ++k) dst[done + k] = detail::be32(t + 4 * k);
827 done += n;
828 }
829 return true;
830 }

832 Error loadHuffTables(uint32_t huffRec, uint32_t huffCount) {
833 if (huffRec == 0 || huffCount < 2 || huffRec >= _numRecords ||
834 huffCount > _numRecords - huffRec) {
835 return fail(Error::CorruptData);
836 }
837 uint32_t start = recordOffset(huffRec), end = recordEnd(huffRec);
838 if (start == kNoRecord || end == kNoRecord || end > _total || end < start ||
839 end - start < 24) {
840 return fail(Error::CorruptData);
841 }
842 uint8_t h[24];
843 if (!readAbs(start, h, sizeof(h))) return fail(Error::Io);
844 if (memcmp(h, "HUFF\0\0\0\x18", 8) != 0) return fail(Error::CorruptData);
845 uint32_t off1 = detail::be32(h + 8), off2 = detail::be32(h + 12);
846 uint32_t recLen = end - start;
847 if (!loadBE32Table(start, off1, recLen, _dict1, 256) ||
848 !loadBE32Table(start, off2, recLen, _dict2, 64)) {
849 return fail(Error::CorruptData);
850 }
851
852 // the CDIC records follow the HUFF record
853 uint32_t cs = recordOffset(huffRec + 1), ce = recordEnd(huffRec + 1);
854 uint8_t c[16];
855 if (cs == kNoRecord || ce == kNoRecord || ce > _total || ce < cs || ce - cs < 16 ||
856 !readAbs(cs, c, sizeof(c)) || memcmp(c, "CDIC\0\0\0\x10", 8) != 0) {
857 return fail(Error::CorruptData);
858 }
859 _phrases = detail::be32(c + 8);
860 _cdicBits = detail::be32(c + 12);
861 if (_cdicBits == 0 || _cdicBits > 16) return fail(Error::CorruptData);
862 _cdicFirst = huffRec + 1;
863 _cdicCount = huffCount - 1;
864 return Error::None;
865 }
866
869 Error huffPhrase(uint32_t r, uint8_t* out, size_t cap, size_t& o, uint32_t depth) {
870 uint32_t cdic = r >> _cdicBits;
871 uint32_t slot = r & ((1u << _cdicBits) - 1);
872 if (cdic >= _cdicCount) return fail(Error::CorruptData);
873 uint32_t rec = _cdicFirst + cdic;
874 uint32_t start = recordOffset(rec), end = recordEnd(rec);
875 if (start == kNoRecord || end == kNoRecord || end > _total || end < start) {
876 return fail(Error::CorruptData);
877 }
878 uint32_t recLen = end - start;
879 uint8_t b2[2];
880 if (16 + 2 * slot + 2 > recLen || !readAbs(start + 16 + 2 * slot, b2, 2)) {
881 return fail(Error::CorruptData);
882 }
883 uint32_t off = detail::be16(b2);
884 if (16 + off + 2 > recLen || !readAbs(start + 16 + off, b2, 2)) {
885 return fail(Error::CorruptData);
886 }
887 uint32_t blen = detail::be16(b2);
888 uint32_t plen = blen & 0x7FFF;
889 uint32_t dataRel = 18 + off;
890 if (dataRel + plen > recLen) return fail(Error::CorruptData);
891 if ((blen & 0x8000) != 0) { // stored as plain text
892 if (o + plen > cap) return fail(Error::BufferTooSmall);
893 if (!readAbs(start + dataRel, out + o, plen)) return fail(Error::Io);
894 o += plen;
895 return Error::None;
896 }
897 return huffUnpack(start + dataRel, plen, out, cap, o, depth + 1);
898 }
899
902 Error huffUnpack(uint32_t srcStart, uint32_t srcLen, uint8_t* out, size_t cap, size_t& o,
903 uint32_t depth) {
904 if (depth > kMaxPhraseDepth) return fail(Error::CorruptData);
905 BitWindow w;
906 w.base = 0;
907 w.len = 0;
908 const uint64_t totalBits = static_cast<uint64_t>(srcLen) * 8;
909 uint64_t bitPos = 0;
910 while (bitPos < totalBits) {
911 uint32_t code = 0;
912 if (!peekBits32(w, srcStart, srcLen, bitPos, code)) return fail(Error::Io);
913 uint32_t v = _dict1[code >> 24];
914 uint32_t codelen = v & 0x1F;
915 if (codelen == 0) return fail(Error::CorruptData);
916 uint64_t maxcode = ((static_cast<uint64_t>(v >> 8) + 1) << (32 - codelen)) - 1;
917 if ((v & 0x80) == 0) { // not terminal in the first table: search the length
918 while (codelen <= 32 && static_cast<uint64_t>(code) < huffMinCode(codelen)) ++codelen;
919 if (codelen > 32) return fail(Error::CorruptData);
920 maxcode = huffMaxCode(codelen);
921 }
922 if (bitPos + codelen > totalBits) break; // padding at the end of the slice
923 bitPos += codelen;
924 if (maxcode < code) return fail(Error::CorruptData);
925 uint64_t r = (maxcode - code) >> (32 - codelen);
926 if (r >= _phrases) return fail(Error::CorruptData);
927 Error e = huffPhrase(static_cast<uint32_t>(r), out, cap, o, depth);
928 if (e != Error::None) return e;
929 }
930 return Error::None;
931 }

1031 static bool parseTagMap(const uint8_t* d, uint32_t len, uint32_t startPos, const TagTable& tt,
1032 TagMap& m) {
1033 memset(&m, 0, sizeof(m));
1034 struct Pending {
1035 uint8_t tag;
1036 uint8_t vpe; // values per group
1037 int32_t count; // number of value groups, or -1 if 'bytes' is used
1038 uint32_t bytes; // size of the value data in bytes
1039 };
1040 Pending pend[kMaxTags];
1041 uint32_t np = 0;
1042 uint32_t cb = 0;
1043 uint32_t dataPos = startPos + tt.controlBytes;
1044 for (uint32_t k = 0; k < tt.count; ++k) {
1045 const TagEntry& te = tt.e[k];
1046 if (te.endFlag == 1) {
1047 ++cb;
1048 continue;
1049 }
1050 if (startPos + cb >= len) return false;
1051 uint8_t mask = te.mask;
1052 uint8_t value = d[startPos + cb] & mask;
1053 if (value == 0) continue;
1054 Pending& pe = pend[np++];
1055 pe.tag = te.tag;
1056 pe.vpe = te.vpe;
1057 pe.bytes = 0;
1058 if (value == mask) {
1059 if (countBits(mask) > 1) {
1060 uint32_t v = 0;
1061 if (!readVwi(d, len, dataPos, v)) return false;
1062 pe.count = -1;
1063 pe.bytes = v;
1064 } else {
1065 pe.count = 1;
1066 }
1067 } else {
1068 while ((mask & 1) == 0) {
1069 mask = static_cast<uint8_t>(mask >> 1);
1070 value = static_cast<uint8_t>(value >> 1);
1071 }
1072 pe.count = value;
1073 }
1074 }
1075 for (uint32_t p = 0; p < np; ++p) {
1076 const Pending& pe = pend[p];
1077 uint32_t idx = 0;
1078 auto take = [&](uint32_t val) {
1079 if (pe.tag < kMaxTags && idx < 2) m.v[pe.tag][idx] = val;
1080 ++idx;
1081 };
1082 if (pe.count >= 0) {
1083 for (int32_t c = 0; c < pe.count; ++c) {
1084 for (uint8_t j = 0; j < pe.vpe; ++j) {
1085 uint32_t v = 0;
1086 if (!readVwi(d, len, dataPos, v)) return false;
1087 take(v);
1088 }
1089 }
1090 } else {
1091 uint32_t consumed = 0;
1092 while (consumed < pe.bytes) {
1093 uint32_t before = dataPos;
1094 uint32_t v = 0;
1095 if (!readVwi(d, len, dataPos, v)) return false;
1096 consumed += dataPos - before;
1097 take(v);
1098 }
1099 }
1100 if (pe.tag < kMaxTags) {
1101 m.has[pe.tag] = true;
1102 m.n[pe.tag] = static_cast<uint8_t>(idx < 2 ? idx : 2);
1103 }
1104 }
1105 return true;
1106 }

1299 g.partDone = false;
1300 g.partFileSet = false;
1301 g.partFile = 0;
1302 g.insertedPart = 0;

1308 bool genNextSegment(Kf8Gen& g, uint32_t& pos, uint32_t& len) {
1309 for (;;) {
1310 if (g.havePending) {
1311 g.havePending = false;
1312 if (g.pendLen == 0) continue;
1313 pos = g.pendPos;
1314 len = g.pendLen;
1315 return true;
1316 }
1317 if (!g.partOpen || g.partDone) return false;
1318 if (g.fragLeft > 0) {
1319 char label[64];
1320 TagMap m;
1321 // the fragment table gives insert positions in the assembled part, so
1322 // the raw skeleton position is shifted back by the fragments inserted so far
1323 uint32_t insertAsm = 0;
1324 if (indxNext(g.frag, label, sizeof(label), m) != Error::None ||
1325 !parseDecimal(label, insertAsm) || !m.has[3] || !m.has[6] || m.n[6] < 2 ||
1326 insertAsm < g.insertedPart) {
1327 g.err = Error::CorruptData;
1328 fail(Error::CorruptData);
1329 return false;
1330 }
1331 uint32_t insert = insertAsm - g.insertedPart;
1332 if (insert < g.skelCur || insert > g.skelEnd) {
1333 g.err = Error::CorruptData;
1334 fail(Error::CorruptData);
1335 return false;
1336 }
1337 if (!g.partFileSet) {
1338 g.partFile = m.v[3][0];
1339 g.partFileSet = true;
1340 }
1341 --g.fragLeft;
1342 g.havePending = true;
1343 g.pendPos = g.baseptr;
1344 g.pendLen = m.v[6][1];
1345 g.baseptr += m.v[6][1];
1346 g.insertedPart += m.v[6][1];
1347 pos = g.skelCur;
1348 len = insert - g.skelCur;
1349 g.skelCur = insert;
1350 if (len > 0) return true;
1351 continue; // deliver the fragment first
1352 }
1353 // the rest of the skeleton ends the part
1354 g.partDone = true;
1355 pos = g.skelCur;
1356 len = g.skelEnd - g.skelCur;
1357 return len > 0;
1358 }
1359 }

1452 size_t rawCopy(uint32_t pos, uint8_t* dst, size_t n) {
1453 if (!WithKf8) return 0;
1454 const uint32_t rs = _kf8View.recordSize;
1455 if (rs == 0) {
1456 fail(Error::CorruptData);
1457 return 0;
1458 }
1459 size_t done = 0;
1460 while (done < n) {
1461 uint32_t off = pos + static_cast<uint32_t>(done);
1462 uint32_t chunk = off / rs;
1463 uint32_t inOff = off - chunk * rs;
1464 if (chunk >= _kf8View.chunks) break;

1469 fail(e);
1470 break;
1471 }

1473 _rawLen = static_cast<uint32_t>(len);
1474 }

1477 if (take > n - done) take = n - done;
1478 memcpy(dst + done, _raw + inOff, take);
1479 done += take;
1480 }
1481 return done;

1518 static bool parseImageTag(const uint8_t* t, size_t n, uint32_t& value) {
1519 if (n < 3 || !ieq(t, "img", 3)) return false;
1520 if (n > 3 && t[3] != ' ' && t[3] != '\t' && t[3] != '\r' && t[3] != '\n' && t[3] != '/') {
1521 return false;
1522 }
1523 for (size_t i = 3; i + 8 <= n; ++i) {
1524 if (!ieq(t + i, "recindex", 8)) continue;
1525 size_t k = i + 8;
1526 while (k < n && t[k] == ' ') ++k;
1527 if (k >= n || t[k] != '=') return false;
1528 ++k;
1529 while (k < n && t[k] == ' ') ++k;
1530 if (k < n && (t[k] == '"' || t[k] == '\'')) ++k;
1531 uint32_t v = 0;
1532 size_t digits = 0;
1533 for (; k < n && t[k] >= '0' && t[k] <= '9' && v < 100000000u; ++k, ++digits) {
1534 v = v * 10 + (t[k] - '0');
1535 }
1536 if (digits == 0) return false;
1537 value = v;
1538 return true;
1539 }
1540 for (size_t i = 3; i + 13 <= n; ++i) {
1541 if (!ieq(t + i, "kindle:embed:", 13)) continue;
1542 size_t k = i + 13;
1543 uint32_t v = 0;
1544 size_t digits = 0;
1545 for (; k < n && v < 0x3FFFFFF; ++k) {
1546 uint8_t c = t[k];
1547 uint32_t d;
1548 if (c >= '0' && c <= '9') d = c - '0';
1549 else if (c >= 'A' && c <= 'V') d = c - 'A' + 10;
1550 else if (c >= 'a' && c <= 'v') d = c - 'a' + 10;
1551 else break;
1552 v = v * 32 + d;
1553 ++digits;
1554 }
1555 if (digits == 0) return false;
1556 value = v;
1557 return true;
1558 }
1559 return false;
1560 }

1852 bool byteFromEnd(ByteWindow& w, uint32_t start, uint32_t len, uint32_t fromEnd, uint8_t& b) {
1853 if (fromEnd == 0 || fromEnd > len) return false;
1854 uint32_t abs = start + len - fromEnd;
1855 if (abs < w.base || abs >= w.base + w.len) {
1856 uint32_t lo = abs >= 15 ? abs - 15 : 0;
1857 if (lo < start) lo = start;

1859 uint32_t n = hi - lo < sizeof(w.buf) ? hi - lo : static_cast<uint32_t>(sizeof(w.buf));
1860 if (!readAbs(lo, w.buf, n)) return false;
1861 w.base = lo;
1862 w.len = n;
1863 }
1864 b = w.buf[abs - w.base];

1867
1870 bool trailingSizeAt(uint32_t start, uint32_t len, uint16_t flags, uint32_t& total) {
1871 ByteWindow w;
1872 uint32_t num = 0;
1873 uint16_t f = static_cast<uint16_t>(flags >> 1);
1874 for (; f != 0; f = static_cast<uint16_t>(f >> 1)) {
1875 if ((f & 1) == 0) continue;
1876 // backward encoded variable length integer; value includes itself
1877 uint32_t value = 0, shift = 0, i = 0;
1878 for (;;) {
1879 uint8_t v = 0;
1880 if (!byteFromEnd(w, start, len, num + i + 1, v)) return false;
1881 ++i;
1882 value |= static_cast<uint32_t>(v & 0x7F) << shift;
1883 shift += 7;
1884 if ((v & 0x80) != 0 || shift >= 28) break;
1885 }
1886 if (value == 0 || value > len - num) return false;
1887 num += value;
1888 }
1889 if ((flags & 1) != 0) { // multibyte overlap descriptor
1890 uint8_t d = 0;
1891 if (!byteFromEnd(w, start, len, num + 1, d)) return false;

1900 Error palmdocDecompress(uint32_t start, uint32_t inLen, uint8_t* out, size_t cap,
1901 size_t& outLen) {
1902 uint8_t in[32];
1903 uint32_t inPos = 0; // bytes fetched from the source
1904 uint32_t inBufLen = 0; // valid bytes in in[]
1905 uint32_t inBufPos = 0;
1906 bool ioError = false;
1907
1908 auto nextByte = [&](uint8_t& b) -> bool {
1909 if (inBufPos >= inBufLen) {
1910 if (inPos >= inLen) return false;
1911 uint32_t want = inLen - inPos < sizeof(in) ? inLen - inPos : sizeof(in);
1912 if (_src.read(start + inPos, in, want) != want) {
1913 ioError = true;
1914 return false;
1915 }
1916 inPos += want;
1917 inBufLen = want;
1918 inBufPos = 0;
1919 }
1920 b = in[inBufPos++];
1921 return true;
1922 };
1923
1924 size_t o = 0;
1925 uint8_t c = 0;
1926 while (nextByte(c)) {
1927 if (c == 0 || (c >= 0x09 && c <= 0x7F)) { // literal
1928 if (o >= cap) return fail(Error::BufferTooSmall);
1929 out[o++] = c;
1930 } else if (c <= 0x08) { // the next c bytes are literal
1931 for (uint8_t k = 0; k < c; ++k) {
1932 uint8_t b = 0;
1933 if (!nextByte(b)) return fail(ioError ? Error::Io : Error::CorruptData);
1934 if (o >= cap) return fail(Error::BufferTooSmall);
1935 out[o++] = b;
1936 }
1937 } else if (c <= 0xBF) { // back reference: 11 bit distance, 3 bit length
1938 uint8_t c2 = 0;
1939 if (!nextByte(c2)) return fail(ioError ? Error::Io : Error::CorruptData);
1940 uint32_t code = ((static_cast<uint32_t>(c) << 8) | c2) & 0x3FFF;
1941 size_t dist = code >> 3;
1942 size_t len = (code & 0x7) + 3;
1943 if (dist == 0 || dist > o) return fail(Error::CorruptData);
1944 if (o + len > cap) return fail(Error::BufferTooSmall);
1945 for (size_t k = 0; k < len; ++k, ++o) out[o] = out[o - dist];
1946 } else { // 0xC0..0xFF: a space followed by a character
1947 if (o + 2 > cap) return fail(Error::BufferTooSmall);
1948 out[o++] = ' ';
1949 out[o++] = static_cast<uint8_t>(c ^ 0x80);
1950 }
1951 }
1952 if (ioError) return fail(Error::Io);
1953 outLen = o;
1954 return Error::None;
1955 }
