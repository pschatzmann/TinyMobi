91 // (nothing is cut off); the image is centred, the rest of the screen is black

136 static bool onCoverBlock(tinyjpeg::TinyJPEGDecoder& decoder, int16_t x, int16_t y, uint16_t w,
137 uint16_t h, uint16_t* data) {
138 MobiCover* self = static_cast<MobiCover*>(decoder.getUserData());
139 if (self == nullptr || !self->cover_) return false;
140 tinygpu::ISurface<RGB_T>& target = *self->cover_;
141 const uint64_t dw = self->decodedW_, dh = self->decodedH_;
142 const uint64_t tw = self->targetW_, th = self->targetH_;
143 for (uint16_t j = 0; j < h; ++j) {
144 const uint64_t dy = static_cast<uint64_t>(y) + j;
145 if (dy >= dh) break;
146 const uint64_t ty0 = dy * th / dh, ty1 = (dy + 1) * th / dh;
147 for (uint16_t i = 0; i < w; ++i) {
148 const uint64_t dx = static_cast<uint64_t>(x) + i;
149 if (dx >= dw) break;
150 const uint64_t tx0 = dx * tw / dw, tx1 = (dx + 1) * tw / dw;
151 const RGB_T color = fromRGB565(data[j * w + i]);
152 for (uint64_t ty = ty0; ty < ty1; ++ty) {
153 for (uint64_t tx = tx0; tx < tx1; ++tx) {
154 const int64_t px = self->offsetX_ + static_cast<int64_t>(tx);
155 const int64_t py = self->offsetY_ + static_cast<int64_t>(ty);
156 if (px >= 0 && py >= 0 && px < static_cast<int64_t>(target.width()) &&
157 py < static_cast<int64_t>(target.height())) {
158 target.setPixel(static_cast<size_t>(px), static_cast<size_t>(py), color);
159 }
160 }
161 }
162 }
163 }
164 return true;
165 }
