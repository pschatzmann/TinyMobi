459 void layoutLines(int32_t width, std::vector<Line>& out) const {
460 out.clear();
461 const std::string& text = this->text();
462 const auto& theme = this->theme();
463 tinygpu::IFont<RGB_T>* body = tinymd::fontForTypographyRole(tinymd::TypographyRole::kBody, theme);
464 tinygpu::IFont<RGB_T>* title = tinymd::fontForTypographyRole(tinymd::TypographyRole::kTitle, theme);
465 size_t ps = 0;
466 for (;;) {
467 size_t pe = text.find('\n', ps);
468 if (pe == std::string::npos) pe = text.size();
469 if (ps == pe && pe == text.size() && !text.empty()) break; // no blank line after the last newline
470 const bool heading = isHeading(ps);
471 wrapParagraph(text, ps, pe, width, heading ? title : body, heading, out);
472 if (pe >= text.size()) break;
473 ps = pe + 1;
474 }
475 int32_t y = this->bounds.y;
476 for (Line& l : out) {
477 l.y = y;
478 l.height = static_cast<int32_t>(l.font->getHeight(1)) + spacing_;
479 y += l.height;
480 }
481 }
482
483 void wrapParagraph(const std::string& text, size_t ps, size_t pe, int32_t width, tinygpu::IFont<RGB_T>* font,
484 bool heading, std::vector<Line>& out) const {
485 const size_t before = out.size();
486 Line line;
487 line.font = font;
488 line.heading = heading;
489 auto fits = [&](const std::string& s) { return static_cast<int32_t>(font->measureTextWidth(s.c_str())) <= width; };
490 auto endLine = [&]() {
491 out.push_back(line);
492 line = Line();
493 line.font = font;
494 line.heading = heading;
495 };
496 size_t wb = std::string::npos;
497 for (size_t i = ps; i <= pe; ++i) {
498 const char c = i < pe ? text[i] : ' ';
499 if (c != ' ') {
500 if (wb == std::string::npos) wb = i;
501 continue;
502 }
503 if (wb == std::string::npos) continue;
504 const std::string word = text.substr(wb, i - wb);
505 const std::string candidate = line.text.empty() ? word : line.text + " " + word;
506 if (!line.text.empty() && !fits(candidate)) {
507 endLine();
508 line.text = word;
509 line.words.push_back(Word{wb, i, 0});
510 } else {
511 line.words.push_back(Word{wb, i, line.text.empty() ? 0 : line.text.size() + 1});
512 line.text = candidate;
513 }
514 wb = std::string::npos;
515 }
516 if (!line.text.empty()) endLine();
517 if (out.size() == before) endLine(); // an empty paragraph is a blank line
518 }

521 bool linkAt(int32_t x, int32_t y, std::string& target) const {
522 if (links_.empty()) return false;
523 std::vector<Line> lines;
524 layoutLines(this->bounds.w, lines);
525 const std::string& text = this->text();
526 for (const Line& l : lines) {
527 // the space between two lines is shared: each line takes half of it on either side
528 const int32_t top = l.y - spacing_ / 2;
529 const int32_t bottom = l.y + l.height - spacing_ / 2;
530 if (y < top || y >= bottom) continue;
531 for (const Word& w : l.words) {
532 const std::string word = text.substr(w.begin, w.end - w.begin);
533 const int32_t x0 = this->bounds.x + static_cast<int32_t>(l.font->measureTextWidth(l.text.substr(0, w.linePos).c_str()));
534 const int32_t x1 = x0 + static_cast<int32_t>(l.font->measureTextWidth(word.c_str()));
535 if (x < x0 || x >= x1) continue;
536 for (const PageLink& k : links_) {
537 if (w.begin >= k.begin && w.begin < k.end) {
538 target = k.target;
539 return true;
540 }
541 }
542 return false;
543 }
544 return false;
545 }
546 return false;
547 }

838 bool readCandidate(Position start, std::string& text) {
839 text.clear();
840 if (doc_.seek(start) != Error::None) return true;
841 uint8_t b = 0;
842 while (text.size() < kPageBytes) {
843 if (doc_.read(&b, 1) == 0) return true;
844 text.push_back(static_cast<char>(b));
845 }
846 return doc_.read(&b, 1) == 0; // true if nothing follows
847 }
848
850 Position endOf(Position start, size_t n) {
851 if (doc_.seek(start) != Error::None) return start;
852 uint8_t b = 0;
853 for (size_t i = 0; i < n; ++i) {
854 if (doc_.read(&b, 1) == 0) break;
855 }
856 return doc_.position();
857 }

860 static bool cutAtWord(std::string& text) {
861 size_t sp = text.find_last_of(" \n\t\r");
862 if (sp == std::string::npos || sp == 0) return false;
863 text.resize(sp);
864 return true;
865 }
866
869 static void stripLinks(const std::string& text, std::string& out, std::vector<PageLink>& links,
870 std::vector<Span>& headings) {
871 out.clear();
872 links.clear();
873 headings.clear();
874 PageLink cur;
875 size_t headBegin = 0;
876 bool inHead = false, inTarget = false, inLink = false;
877 for (char ch : text) {
878 const uint8_t b = static_cast<uint8_t>(ch);
879 if (b == kHeadBegin) {
880 headBegin = out.size();
881 inHead = true;
882 } else if (b == kHeadEnd) {
883 if (inHead && out.size() > headBegin) headings.push_back(Span{headBegin, out.size()});
884 inHead = false;
885 } else if (b == kLinkBegin) {
886 cur = PageLink();
887 cur.begin = out.size();
888 cur.target.clear();
889 inTarget = inLink = true;
890 } else if (inTarget) {
891 if (b == kLinkTarget) inTarget = false;
892 else cur.target.push_back(ch);
893 } else if (b == kLinkEnd) {
894 if (inLink && cur.end > cur.begin) links.push_back(cur);
895 inLink = false;
896 } else {
897 out.push_back(ch);
898 if (inLink) cur.end = out.size();
899 }
900 }
901 // a link or heading cut by the end of the page: keep its visible part
902 if (inLink && !inTarget && cur.end > cur.begin) links.push_back(cur);
903 if (inHead && out.size() > headBegin) headings.push_back(Span{headBegin, out.size()});
904 }
905
908 bool fitText(std::string& text) {
909 bool cut = false;
910 std::string shown;
911 std::vector<PageLink> links;
912 std::vector<Span> headings;
913 // the text block gets the headings too: its height depends on them
914 stripLinks(text, shown, links, headings);
915 text_.setPage(shown, links, headings);
916 while (!text.empty() && text_.textHeight(text_.bounds.w) > text_.bounds.h) {
917 if (text.size() < 2) break;
918 size_t sp = text.find_last_of(" \n\t\r", text.size() - 2);
919 if (sp == std::string::npos || sp == 0) break;
920 text.resize(sp);
921 cut = true;
922 stripLinks(text, shown, links, headings);
923 text_.setPage(shown, links, headings);
924 }
925 return cut;
926 }
