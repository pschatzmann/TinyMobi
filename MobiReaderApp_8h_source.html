478 void layoutLines(int32_t width, std::vector<Line>& out) const {
479 out.clear();
480 const std::string& text = this->text();
481 const auto& theme = this->theme();
482 tinygpu::IFont<RGB_T>* body = tinymd::fontForTypographyRole(tinymd::TypographyRole::kBody, theme);
483 tinygpu::IFont<RGB_T>* title = tinymd::fontForTypographyRole(tinymd::TypographyRole::kTitle, theme);
484 size_t ps = 0;
485 for (;;) {
486 size_t pe = text.find('\n', ps);
487 if (pe == std::string::npos) pe = text.size();
488 if (ps == pe && pe == text.size() && !text.empty()) break; // no blank line after the last newline
489 const bool heading = isHeading(ps);
490 wrapParagraph(text, ps, pe, width, heading ? title : body, heading, out);
491 if (pe >= text.size()) break;
492 ps = pe + 1;
493 }
494 int32_t y = this->bounds.y;
495 for (Line& l : out) {
496 l.y = y;
497 l.height = static_cast<int32_t>(l.font->getHeight(1)) + spacing_;
498 y += l.height;
499 }
500 }
501
502 void wrapParagraph(const std::string& text, size_t ps, size_t pe, int32_t width, tinygpu::IFont<RGB_T>* font,
503 bool heading, std::vector<Line>& out) const {
504 const size_t before = out.size();
505 Line line;
506 line.font = font;
507 line.heading = heading;
508 auto fits = [&](const std::string& s) { return static_cast<int32_t>(font->measureTextWidth(s.c_str())) <= width; };
509 auto endLine = [&]() {
510 out.push_back(line);
511 line = Line();
512 line.font = font;
513 line.heading = heading;
514 };
515 size_t wb = std::string::npos;
516 for (size_t i = ps; i <= pe; ++i) {
517 const char c = i < pe ? text[i] : ' ';
518 if (c != ' ') {
519 if (wb == std::string::npos) wb = i;
520 continue;
521 }
522 if (wb == std::string::npos) continue;
523 const std::string word = text.substr(wb, i - wb);
524 const std::string candidate = line.text.empty() ? word : line.text + " " + word;
525 if (!line.text.empty() && !fits(candidate)) {
526 endLine();
527 line.text = word;
528 line.words.push_back(Word{wb, i, 0});
529 } else {
530 line.words.push_back(Word{wb, i, line.text.empty() ? 0 : line.text.size() + 1});
531 line.text = candidate;
532 }
533 wb = std::string::npos;
534 }
535 if (!line.text.empty()) endLine();
536 if (out.size() == before) endLine(); // an empty paragraph is a blank line
537 }

540 bool linkAt(int32_t x, int32_t y, std::string& target) const {
541 if (links_.empty()) return false;
542 std::vector<Line> lines;
543 layoutLines(this->bounds.w, lines);
544 const std::string& text = this->text();
545 for (const Line& l : lines) {
546 // the space between two lines is shared: each line takes half of it on either side
547 const int32_t top = l.y - spacing_ / 2;
548 const int32_t bottom = l.y + l.height - spacing_ / 2;
549 if (y < top || y >= bottom) continue;
550 for (const Word& w : l.words) {
551 const std::string word = text.substr(w.begin, w.end - w.begin);
552 const int32_t x0 = this->bounds.x + static_cast<int32_t>(l.font->measureTextWidth(l.text.substr(0, w.linePos).c_str()));
553 const int32_t x1 = x0 + static_cast<int32_t>(l.font->measureTextWidth(word.c_str()));
554 if (x < x0 || x >= x1) continue;
555 for (const PageLink& k : links_) {
556 if (w.begin >= k.begin && w.begin < k.end) {
557 target = k.target;
558 return true;
559 }
560 }
561 return false;
562 }
563 return false;
564 }
565 return false;
566 }

872 bool readCandidate(Position start, std::string& text) {
873 text.clear();
874 if (doc_.seek(start) != Error::None) return true;
875 uint8_t b = 0;
876 while (text.size() < kPageBytes) {
877 if (doc_.read(&b, 1) == 0) return true;
878 text.push_back(static_cast<char>(b));
879 }
880 return doc_.read(&b, 1) == 0; // true if nothing follows
881 }
882
884 Position endOf(Position start, size_t n) {
885 if (doc_.seek(start) != Error::None) return start;
886 uint8_t b = 0;
887 for (size_t i = 0; i < n; ++i) {
888 if (doc_.read(&b, 1) == 0) break;
889 }
890 return doc_.position();
891 }

894 static bool cutAtWord(std::string& text) {
895 size_t sp = text.find_last_of(" \n\t\r");
896 if (sp == std::string::npos || sp == 0) return false;
897 text.resize(sp);
898 return true;
899 }

907 static void stripLinks(const std::string& text, std::string& out, std::vector<PageLink>& links,
908 std::vector<Span>& headings, const std::string& hiddenHeading) {
909 out.clear();
910 links.clear();
911 headings.clear();
912 PageLink cur;
913 size_t headBegin = 0;
914 bool inHead = false, inTarget = false, inLink = false;
915 for (char ch : text) {
916 const uint8_t b = static_cast<uint8_t>(ch);
917 if (b == kHeadBegin) {
918 headBegin = out.size();
919 inHead = true;
920 } else if (b == kHeadEnd) {
921 if (inHead && out.size() > headBegin) headings.push_back(Span{headBegin, out.size()});
922 inHead = false;
923 } else if (b == kLinkBegin) {
924 cur = PageLink();
925 cur.begin = out.size();
926 cur.target.clear();
927 inTarget = inLink = true;
928 } else if (inTarget) {
929 if (b == kLinkTarget) inTarget = false;
930 else cur.target.push_back(ch);
931 } else if (b == kLinkEnd) {
932 if (inLink && cur.end > cur.begin) links.push_back(cur);
933 inLink = false;
934 } else {
935 if (b == '\n' && out.empty()) continue; // no empty line at the top of the page
936 out.push_back(ch);
937 if (inLink) cur.end = out.size();
938 }
939 }
940 // a link or heading cut by the end of the page: keep its visible part
941 if (inLink && !inTarget && cur.end > cur.begin) links.push_back(cur);
942 if (inHead && out.size() > headBegin) headings.push_back(Span{headBegin, out.size()});
943
944 // a first heading that is the book title is not shown again (the title bar has it)
945 if (!headings.empty() && headings[0].begin == 0 && !hiddenHeading.empty() &&
946 trimmed(out.substr(0, headings[0].end)) == trimmed(hiddenHeading)) {
947 size_t cut = headings[0].end;
948 while (cut < out.size() && out[cut] == '\n') ++cut;
949 out.erase(0, cut);
950 headings.erase(headings.begin());
951 for (Span& h : headings) {
952 h.begin -= cut;
953 h.end -= cut;
954 }
955 std::vector<PageLink> kept;
956 for (PageLink& l : links) {
957 if (l.begin < cut) continue;
958 l.begin -= cut;
959 l.end -= cut;
960 kept.push_back(l);
961 }
962 links.swap(kept);
963 }
964 }

966 static std::string trimmed(const std::string& t) {
967 const size_t b = t.find_first_not_of(" \n\t\r");
968 if (b == std::string::npos) return std::string();
969 const size_t e = t.find_last_not_of(" \n\t\r");
970 return t.substr(b, e - b + 1);
971 }

975 bool fitText(std::string& text) {
976 bool cut = false;
977 std::string shown;
978 std::vector<PageLink> links;
979 std::vector<Span> headings;
980 // the text block gets the headings too: its height depends on them
981 stripLinks(text, shown, links, headings, bookTitle_);
982 text_.setPage(shown, links, headings);
983 while (!text.empty() && text_.textHeight(text_.bounds.w) > text_.bounds.h) {
984 if (text.size() < 2) break;
985 size_t sp = text.find_last_of(" \n\t\r", text.size() - 2);
986 if (sp == std::string::npos || sp == 0) break;
987 text.resize(sp);
988 cut = true;
989 stripLinks(text, shown, links, headings, bookTitle_);
990 text_.setPage(shown, links, headings);
991 }
992 return cut;
993 }
